{# Generic record field grid, driven by dataset.layout. Used by the record view/edit and the add-new form for every archive, so field order, labels and widths live in the registry (datasets.py) rather than in a template per archive. A row holds cells: either a single Field, or a Stack of fields sharing one column. Columns in a row are equal height, so a field marked grow=True stretches to fill whatever its stack does not use. #} {% macro textfield(name, label, value, readonly=True) %} {% endmacro %} {% macro areafield(name, label, value, rows, readonly=True) %} {% endmacro %} {% macro field(dataset, record, f, readonly) %} {% if f.widget == 'textarea' %} {{ areafield(f.attr, dataset.label(f.attr), record[f.attr], f.rows, readonly) }} {% else %} {{ textfield(f.attr, dataset.label(f.attr), record[f.attr], readonly) }} {% endif %} {% endmacro %} {% macro record_fields(dataset, record, editable=False) %} {% set ro = not editable %} {% for row in dataset.layout %}
{% for cell in row %}
{% if cell.is_stack %}
{% for f in cell.fields %}
{{ field(dataset, record, f, ro) }}
{% endfor %}
{% else %} {{ field(dataset, record, cell, ro) }} {% endif %}
{% endfor %}
{% endfor %} {% endmacro %}
